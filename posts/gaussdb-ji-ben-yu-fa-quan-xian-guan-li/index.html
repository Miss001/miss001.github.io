GaussDB 基本语法：权限管理

要点：创建普通用户、授予管理员权限、授予表权限、授予行级权限。

宇宙超级无敌美少女数据库国产数据库GaussDB

创建普通用户#

CREATE USER alice PASSWORD '<PASSWORD>';

授予管理员权限#

# 授予管理员权限
GRANT ALL PRIVILEGES TO alice;

授予表权限#

GRANT USAGE ON SCHEMA root TO alice;
GRANT SELECT ON TABLE root.all_data to alice;

授予行级权限#

ALTER TABLE all_data ENABLE ROW LEVEL SECURITY
CREATE ROW LEVEL SECURITY POLICY all_data_rls ON all_data USING(role = CURRENT_USER);

# 切换用户alice查询数据
select * from root.all_data
 id | role | data
----+-------+------------
 1 | alice | alice data
(1 row)
