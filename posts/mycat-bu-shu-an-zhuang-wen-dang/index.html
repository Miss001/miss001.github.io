MyCat 部署：安装文档

宇宙超级无敌美少女数据库MyCat

1）下载

下载地址：https://github.com/MyCATApache/Mycat-Server/releases

2）安装

#解压
tar -xvf Mycat-server-1.6.7.5-release-20200422133810-linux.tar.gz

#添加用户
adduser mycat
chown mycat:mycat mycat/

#配置环境变量
vi /etc/profile
MYCAT_HOME=/opt/mycat/mycat
export PATH=$PATH:$MYCAT_HOME/bin

3）配置

引用链接：Mycat 核心配置详解-腾讯云开发者社区-腾讯云 (tencent.com)

schema.xml

用途：

  • 配置逻辑库及逻辑表
  • 配置逻辑表所存储的数据节点
  • 配置数据节点所对应的物理数据库服务器信息
<?xml version="1.0"?>
<!DOCTYPE mycat:schema SYSTEM "schema.dtd">
<mycat:schema xmlns:mycat="http://io.mycat/">

        <!-- 定义mycat中的逻辑库 -->
        <schema name="TESTDB" checkSQLschema="true" sqlMaxLimit="100" randomDataNode="dn1">
          <!-- 定义mycat中的逻辑表 -->
          <!-- 多表定义 -->
          <table name="tmp_demo,tmp_demo1" dataNode="dn1,dn2" rule="auto-sharding-long" splitTableNames ="true"/>
          <!-- 单表定义 -->
          <!-- <table name="oc_call" primaryKey="id" dataNode="dn1$0-743" rule="latest-month-calldate"/> -->
        </schema>


        <!--定义数据节点，数据节点指向的是存储逻辑表的物理数据库-->
        <dataNode name="dn1" dataHost="localhost1" database="ceshi" />
        <dataNode name="dn2" dataHost="localhost1" database="ceshi" />
        <!-- 可以配置一个范围 -->
        <!-- <dataNode name="dn1$0-743" dataHost="localhost1" database="db$0-743"/> -->


        <!--定义后端物理数据库主机信息-->
        <dataHost name="localhost1" maxCon="1000" minCon="10" balance="0"
                  writeType="0" dbType="mysql" dbDriver="jdbc" switchType="1" slaveThreshold="100">
                <heartbeat>select user()</heartbeat>
                <!-- 多个实例配置 -->
                <writeHost host="localhost" url="jdbc:mysql://localhost:31301" user="root" password="<PASSWORD>">
                   <readHost host="localhost" url="jdbc:mysql://localhost:31302" user="root" password="<PASSWORD>"></readHost>
                </writeHost>
        </dataHost>
</mycat:schema>

rule.xml

  • 配置水平分片的分片规则
  • 配置分片规则所对应的分片函数

默认文件即可使用

4）替换lib/mysql-connector-java-5.1.35.jar为8的版本

5）启动
