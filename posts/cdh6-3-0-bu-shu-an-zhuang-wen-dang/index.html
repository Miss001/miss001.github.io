二 环境准备#

设置机器名称#

  1. 修改主机名
hostnamectl set-hostname host1

hostnamectl set-hostname host2

hostnamectl set-hostname host3

hostnamectl set-hostname host4

hostnamectl set-hostname host5
  1. 设置映射文件

vi /etc/hosts

ip  host1
ip  host2
ip  host3
ip  host4
ip  host5

设置SSH免密登录#

  1. 去掉/etc/ssh/sshd_config的两行注释

所有机器执行

RSAAuthentication yes
PubkeyAuthentication yes
  1. 生成公钥私钥

所有机器执行

输入命令：ssh-keygen -t rsa，一直回车，/root就会生成.ssh文件夹（ls看不到）

id_rsa（私钥）id_rsa.pub（公钥）

  1. 将主节点公钥传至其他节点

主节点执行

cat id_rsa.pub>> authorized_keys
scp -P22 -r authorized_keys  root@host2:~/.ssh/authorized_keys
scp -P22 -r authorized_keys  root@host3:~/.ssh/authorized_keys
scp -P22 -r authorized_keys  root@host4:~/.ssh/authorized_keys
scp -P22 -r authorized_keys  root@host5:~/.ssh/authorized_keys

磁盘目录挂载#

lsblk：查看磁盘和分区

所有机器执行

脚本挂载：

sh mount_disk.sh  /dev/sda /data/1

脚本见本仓库 ops/linux/磁盘挂载/mount_disk.sh。

设置集群时间同步#

#下载
yum -y install chrony

#修改配置

#开机自启动
systemctl enable chronyd

systemctl start  chronyd

#开启同步
chronyc sources

配置yum仓库#

yum源创建方法:

  1. 安装 Http 服务器软件
yum -y install httpd createrepo
  1. 启动 http 服务并设置开机自启动
systemctl start httpd
systemctl enable httpd
  1. 创建安装包目录上传安装包
mkdir -p /var/www/html/cloudera-repos

上传所有安装包至该目录
  1. 创建 Yum 仓库的 Http 目录
cd /var/www/html/cloudera-repos
createrepo .
  1. 将安装包上传到/var/www/html/cloudera-repos目录下

在所有机器上增加配置

vi /etc/yum.repos.d/cloudera-manager.repo

#增加内容
[cloudera-manager]
name=Cloudera Manager 6.3.0
baseurl=http://host1/cloudera-repos/
gpgcheck=0
enabled=1

系统环境配置#

  • 交换空间swapping设置
cat /proc/sys/vm/swappiness
sysctl -w vm.swappiness=0
  • 修改openfiles

vi /etc/security/limits.conf

*  soft  nofile  655360
*  hard  nofile  655360

三 部署#

安装jdk#

  1. 上传安装包
jdk-8u191-linux-x64_.tar.gz
  1. 解压安装包
tar -zxvf jdk-8u191-linux-x64_.tar.gz
  1. 移动文件至指定目录

一定要移到 /usr/java 目录下，否则启动角色会报错

mkdir /usr/java
mv /home/jdk1.8.0_191 /usr/java/jdk1.8
  1. 配置环境变量

vi /etc/profile

export JAVA_HOME=/usr/java/jdk1.8
export PATH=$JAVA_HOME/bin:$PATH

source /etc/profile

  1. 验证
javac
java -version

安装mysql#

<!--root用户 host1节点安装-->
  1. 检查环境删除mariadb包
rpm -qa |grep mariadb
rpm -e --nodeps mariadb-libs-5.5.68-1.el7.x86_64
  1. 上传安装包
mysql-8.0.21-linux-glibc2.12-x86_64.tar.xz
  1. 解压安装
tar -Jxf mysql-8.0.21-linux-glibc2.12-x86_64.tar.xz
mv mysql-8.0.21-linux-glibc2.12-x86_64 /home/mysql/mysql
mkdir -p /home/mysql/data
  1. 配置my.cnf
[mysqld]
# 设置3306端口
port=3306
character-set-server=utf8mb4
default-storage-engine=INNODB
default_authentication_plugin=mysql_native_password
lower_case_table_names=1
innodb_file_per_table=1

basedir=/home/mysql/mysql
datadir=/home/mysql/mysqldata

log-error = /home/mysql/mysqldata/error.log  #错误日志路径（docker配置修改此路径为其他目录需要修改容器内文件权限）
slow-query-log = ON  #慢查询开启开关
long_query_time = 2  #记录大于2秒的SQL语句。
log_queries_not_using_indexes  #没有使用到索引的SQL语句。
slow-query-log-file =/home/mysql/mysqldata/slow.log  #慢日志路径（docker配置修改此路径为其他目录需要修改容器内文件权限）
#min_examined_row_limit = 800  #记录结果集大于800行的SQL语句。

#开启二进制日志
binlog_expire_logs_seconds=2073600
server_id = 1   #服务器id，唯一
log-bin=mysql-bin  #binlog文件名，默认就可以
max_binlog_size = 1G  #binlog单个文件大小，可调节（最小值为4096字节，最大值和默认值为1GB）
binlog_format = row  #行模式
binlog_row_image = minimal  #full （记录所有列）；minimal （仅记录更改的列和标识行所需的列）；noblob （记录所有列，除了不需要的BLOB和TEXT列）
binlog_rows_query_log_events=1 #默认关闭，启用后，它会使服务器将信息日志事件（如行查询日志事件）写入其二进制日志中
innodb_flush_log_at_trx_commit=2
sync_binlog=1
max_connections=1000

interactive_timeout=1800
wait_timeout=1800
max_allowed_packet= 64M

sort_buffer_size=256K
join_buffer_size=256K
binlog_cache_size=2M

innodb_buffer_pool_size=128M
innodb_buffer_pool_instances=1
innodb_thread_concurrency=0

[mysql]
[client]
port=3306
  1. 用户授权
chown -R mysql:mysql /home/mysql
  1. 初始化mysql
su mysql

cd /home/mysql/mysql

./bin/mysqld --defaults-file=/etc/my.cnf --initialize  --user=mysql --datadir=/home/mysql/mysqldata --basedir=/home/mysql/mysql

exit
  1. 启动mysql
cp /home/mysql/mysql/support-files/mysql.server  /etc/init.d/mysqld
service mysqld start
ln -s /home/mysql/mysql/bin/mysql /usr/bin
  1. 修改登陆密码
#获取临时密码 A temporary password is generated for root@localhost: <PASSWORD>
tail -1000f /home/mysql/data/error.log 
#登陆mysql
mysql -uroot -p
#修改当前用户密码
alter user user() identified by '<PASSWORD>';
#修改用户远程访问权限
use mysql;
update user set host = '%' where user = 'root';
flush privileges;
#授权
grant all privileges  on *.*  to 'root'@'%' with grant option;
#修改密码
ALTER USER 'root'@'%' IDENTIFIED WITH mysql_native_password BY '<PASSWORD>';
#刷新
flush privileges;
  1. 创建数据库
CREATE DATABASE cdh_scm DEFAULT CHARACTER SET utf8 DEFAULT COLLATE utf8_general_ci;
CREATE DATABASE cdh_hue DEFAULT CHARACTER SET utf8 DEFAULT COLLATE utf8_general_ci;
CREATE DATABASE cdh_hive DEFAULT CHARACTER SET utf8 DEFAULT COLLATE utf8_general_ci;
CREATE DATABASE cdh_repos DEFAULT CHARACTER SET utf8 DEFAULT COLLATE utf8_general_ci;
CREATE DATABASE cdh_rm DEFAULT CHARACTER SET utf8 DEFAULT COLLATE utf8_general_ci;
CREATE DATABASE cdh_sentry DEFAULT CHARACTER SET utf8 DEFAULT COLLATE utf8_general_ci;
  1. 开机自启动
systemctl enable mysqld.service

service mysqld start

  1. 创建安装目录
 mkdir -p /opt/cloudera/

yum -y install perl
  1. 上传manager 安装包
cd /opt/cloudera/
cloudera-manager-daemons-6.3.0-1281944.el7.x86_64.rpm

rpm -ivh cloudera-manager-daemons-6.3.0-1281944.el7.x86_64.rpm
rpm -ivh cloudera-manager-server-6.3.0-1281944.el7.x86_64.rpm

cp mysql-connector-java-5.1.46.jar  /usr/share/java/mysql-connector-java.jar

/opt/cloudera/cm/schema/scm_prepare_database.sh -h host1 -P 3306 --scm-host host1  mysql cdh_scm root <PASSWORD>
