Elasticsearch：DSL查询

要点：range范围查询、script范围查询。

宇宙超级无敌美少女大数据Elasticsearch

range范围查询#

字段类型为：
integer_range
float_range
long_range
double_range
date_range
ip_range 

{
    "query": {
        "bool": {
            "must": [
                                 {"range":{"num":{"gte":10,"lte":100}}}
                ]
            }
        }
    }

{
    "query": {
        "bool": {
            "must": [
                                 {"range":{"timestamp" : {
                                "gte": "2015-01-01 00:00:00", 
                                "lt": "2016-01-01 00:00:00",
                                                "format": "yyyy-MM-dd HH:mm:ss"
                                "time_zone": "+01:00"
                            }
                    }
                   }
                ]
            }
        }
    }

script范围查询#

字段类型为: keyword

{
    "query": {
        "bool": {
            "must": [{
                    "script": {
                        "script": "
                        if (doc['je'].size() > 0 && doc['je'].value != \"\" && Float.parseFloat(doc['je'].value)>=100 && Float.parseFloat(doc['je'].value)<=500) {return true}
                            else {return false}
                            "
                        }
                    }
                ]
            }
        }
    }

字段类型为: date

{
    "query": {
        "bool": {
            "must": [{
                    "script": {
                        "script": "
                        if (doc['kssj'].size() == 0 ) {return new Date().getTime()> doc['jssj'].value.getMillis()}
                        else {return doc['kssj'].value.getMillis() > doc['jssj'].value.getMillis()}
                        "
                        }
                    }
                ]
            }
        }
    }
